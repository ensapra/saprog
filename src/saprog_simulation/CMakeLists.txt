cmake_minimum_required(VERSION 3.8)
project(saprog_simulation)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(saprog_sim_actuators SHARED src/actuators.cpp)
target_compile_features(saprog_sim_actuators PRIVATE cxx_std_17)
target_link_libraries(saprog_sim_actuators gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS saprog_sim_actuators LIBRARY DESTINATION lib)
install(DIRECTORY launch worlds DESTINATION share/${PROJECT_NAME})
ament_package()
